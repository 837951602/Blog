编译结果
    
    imul rdi, rsi     ; 读入ull可以避免这里的显式截断   ; 3cy
    movabs rax, -6148914695054265911                ;
    mul rdi                                         ; 4cy for rdx
    shr rdx, 31                                     ; 1cy
    mov eax, edx      ; 0xc0000001 = 1 - 0x40000000 
    shl eax, 30       ; 因此甚至不需要一次lea来*3      ; 1cy
    sub edi, edx      ; 私改，本来是sub eax
    add eax, edi                                    ; 1cy
    ret                                         ; sum 10cy
